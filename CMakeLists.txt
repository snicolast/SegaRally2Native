# Ninja learns each file's headers from cl's /showIncludes lines. With a
# Visual Studio in another language they are translated ("Nota: inclusión
# del archivo:"), and the accented prefix CMake had detected did not match
# what cl printed in the console's code page: no header was tracked, and a
# changed header left stale objects behind (classes of two layouts in one
# program). build.bat runs everything with the console in UTF-8 (chcp
# 65001); the prefix is taken here from cl itself, in UTF-8 too
if(MSVC AND CMAKE_GENERATOR MATCHES "Ninja")
  file(WRITE "${CMAKE_BINARY_DIR}/showincludes.c" "#include <stddef.h>\n")
  execute_process(COMMAND "${CMAKE_C_COMPILER}" /nologo /showIncludes /Zs showincludes.c
                  WORKING_DIRECTORY "${CMAKE_BINARY_DIR}" OUTPUT_VARIABLE _incl ERROR_VARIABLE _incl ENCODING UTF8)
  if(_incl MATCHES "\n([^\n]*[^ \n]) *[A-Za-z]:\\\\[^\n]*stddef\\.h")
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "${CMAKE_MATCH_1} ")
  endif()
endif()

# glslang's precompiled header (cmake_pch.hxx) is written in UTF-8 with the
# full path of pch.h: without /utf-8, cl read it in the ANSI code page and a
# folder with accents (C:\Users\José\...) came out mangled (C1083)
if(MSVC)
    add_compile_options($<$<COMPILE_LANGUAGE:C,CXX>:/utf-8>)
endif()

set(SHADER_DIR ${CMAKE_SOURCE_DIR}/src/video/shaders)
set(SHADER_OUT ${CMAKE_BINARY_DIR}/shaders)
file(GLOB SHADER_SOURCES ${SHADER_DIR}/*.vert ${SHADER_DIR}/*.frag ${SHADER_DIR}/*.comp)
set(SHADER_HEADERS)
foreach(src ${SHADER_SOURCES})
    get_filename_component(name ${src} NAME)
    string(REPLACE "." "_" var ${name})
    set(out ${SHADER_OUT}/${name}.h)
    # glslang writes the files each shader #includes to ${out}.d, so a change
    # in any of them rebuilds it (no list of includes to keep by hand)
    add_custom_command(OUTPUT ${out}
        COMMAND ${CMAKE_COMMAND} -E make_directory ${SHADER_OUT}
        COMMAND $<TARGET_FILE:glslang-standalone> -V --target-env vulkan1.2 --vn spv_${var} --depfile ${out}.d -o ${out} ${src}
        DEPENDS ${src} glslang-standalone
        DEPFILE ${out}.d
        COMMENT "Shader ${name}")
    list(APPEND SHADER_HEADERS ${out})
endforeach()
# Again with the ray traced effects (sun shadows, ambient occlusion), for GPUs
# with ray queries: r3d_rt.frag.h, ao_rt.frag.h, car_probe_rt.frag.h
foreach(name r3d ao car_probe)
    set(out ${SHADER_OUT}/${name}_rt.frag.h)
    add_custom_command(OUTPUT ${out}
        COMMAND ${CMAKE_COMMAND} -E make_directory ${SHADER_OUT}
        COMMAND $<TARGET_FILE:glslang-standalone> -V --target-env vulkan1.2 -DRAY_TRACING --vn spv_${name}_rt_frag --depfile ${out}.d -o ${out} ${SHADER_DIR}/${name}.frag
        DEPENDS ${SHADER_DIR}/${name}.frag glslang-standalone
        DEPFILE ${out}.d
        COMMENT "Shader ${name}.frag (ray tracing)")
    list(APPEND SHADER_HEADERS ${out})
endforeach()
# ... and the 3D pass's vertex shader (the light of the dust, traced for each puff): r3d_rt.vert.h
set(out ${SHADER_OUT}/r3d_rt.vert.h)
add_custom_command(OUTPUT ${out}
    COMMAND ${CMAKE_COMMAND} -E make_directory ${SHADER_OUT}
    COMMAND $<TARGET_FILE:glslang-standalone> -V --target-env vulkan1.2 -DRAY_TRACING --vn spv_r3d_rt_vert --depfile ${out}.d -o ${out} ${SHADER_DIR}/r3d.vert
    DEPENDS ${SHADER_DIR}/r3d.vert glslang-standalone
    DEPFILE ${out}.d
    COMMENT "Shader r3d.vert (ray tracing)")
list(APPEND SHADER_HEADERS ${out})
add_custom_target(shaders DEPENDS ${SHADER_HEADERS})

# SDL3, for the gamepads only (no window, sound or rendering of its own):
# downloaded once at configure time and linked statically
include(FetchContent)
# SDL3 and miniz (third_party/deps.txt: file, URL, SHA256). build.bat downloads
# them first into third_party/downloads (tools/get_deps.bat, with retries); a
# copy there is used instead of the URL, so a file saved there by hand works too
function(sr2_dep name)
    file(STRINGS ${CMAKE_SOURCE_DIR}/third_party/deps.txt line REGEX "^${name} ")
    string(STRIP "${line}" line)
    string(REGEX REPLACE "[ \t]+" ";" line "${line}")
    list(GET line 1 file)
    list(GET line 2 url)
    list(GET line 3 hash)
    if(EXISTS ${CMAKE_SOURCE_DIR}/third_party/downloads/${file})
        set(url ${CMAKE_SOURCE_DIR}/third_party/downloads/${file})
    endif()
    set(${name}_URL ${url} PARENT_SCOPE)
    set(${name}_HASH SHA256=${hash} PARENT_SCOPE)
endfunction()

# SDL looks for DirectInput only with its audio or video on (both off above).
# MSVC gets it anyway (SDL presets its checks there); clang or GCC for Windows
# (llvm-mingw: NicoLauncher's builds) did not, and without it the wheels that
# only speak DirectInput (Fanatec...) were not seen and no wheel had force
# feedback (issue #33)
if(WIN32 AND NOT MSVC)
    set(HAVE_DINPUT_H 1 CACHE INTERNAL "Have include dinput.h")
endif()
sr2_dep(SDL3)
FetchContent_Declare(SDL3
    URL ${SDL3_URL}
    URL_HASH ${SDL3_HASH}
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE
    EXCLUDE_FROM_ALL)
FetchContent_MakeAvailable(SDL3)

# NVIDIA DLSS SDK (tools/get_dlss.bat fetches it): optional. Without it the game
# builds with a stub, and --aa dlaa / dlss fall back to MSAA as on a GPU without DLSS.
# The SDK is a static library for MSVC and links with nothing else: any other
# compiler on Windows builds the game with dlss_loader.cpp, which looks for
# dlss_bridge.dll (tools/dlss_bridge) next to the executable. DLSS_BRIDGE=ON
# does the same with MSVC.
if(WIN32 AND NOT MSVC)
    set(DLSS_BRIDGE_DEFAULT ON)
else()
    set(DLSS_BRIDGE_DEFAULT OFF)
endif()
option(DLSS_BRIDGE "DLAA and DLSS through dlss_bridge.dll instead of linking the NVIDIA DLSS SDK into the game" ${DLSS_BRIDGE_DEFAULT})
set(NGX_FILES lib/nvsdk_ngx_d.lib include/nvsdk_ngx.h include/nvsdk_ngx_defs.h include/nvsdk_ngx_defs_vk.h
    include/nvsdk_ngx_helpers.h include/nvsdk_ngx_helpers_vk.h include/nvsdk_ngx_loader.h include/nvsdk_ngx_params.h include/nvsdk_ngx_vk.h)
set(NGX_OK TRUE)
foreach(f ${NGX_FILES})
    if(NOT EXISTS ${CMAKE_SOURCE_DIR}/third_party/ngx/${f})
        set(NGX_OK FALSE)
    endif()
endforeach()
if(WIN32 AND (DLSS_BRIDGE OR NOT MSVC))
    set(DLSS_SOURCE src/video/dlss_loader.cpp)
    set(DLSS_LIB "")
elseif(NGX_OK)
    set(DLSS_SOURCE src/video/dlss.cpp)
    set(DLSS_LIB ${CMAKE_SOURCE_DIR}/third_party/ngx/lib/nvsdk_ngx_d.lib)
else()
    message(STATUS "NVIDIA DLSS SDK not found in third_party/ngx: building without DLAA/DLSS")
    set(DLSS_SOURCE src/video/dlss_stub.cpp)
    set(DLSS_LIB "")
endif()

# The achievement icons (res/achievements/*.png) built into the exe as byte
# arrays (achievement_icons.h): the game writes them next to
# achievements/achievements.csv (src/runtime/achievements_csv.cpp)
set(ICONS_DIR ${CMAKE_CURRENT_BINARY_DIR}/achievement_icons)
set(icons_h "")
foreach(icon unlocked locked)
    set(png ${CMAKE_CURRENT_SOURCE_DIR}/res/achievements/${icon}.png)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${png})
    file(READ ${png} hex HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," hex "${hex}")
    string(APPEND icons_h "static const unsigned char ICON_${icon}[] = {${hex}};\n")
endforeach()
file(CONFIGURE OUTPUT ${ICONS_DIR}/achievement_icons.h CONTENT "${icons_h}")
target_include_directories(SegaRally2 PRIVATE ${ICONS_DIR})
target_link_libraries(SegaRally2 PRIVATE sr2_video SDL3::SDL3-static winmm ole32 oleaut32 windowscodecs mfplat mfuuid gdi32 dwmapi shell32 ws2_32
    ${DLSS_LIB})

if(MSVC)
    # /bigobj: the generated files contain many functions. /W4 on our C++ (the
    # C of Musashi and volk keeps the default level)
    target_compile_options(SegaRally2 PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/bigobj /W4 /wd4146 /wd4102 /fp:strict /utf-8>)
    target_compile_definitions(SegaRally2 PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_link_options(SegaRally2 PRIVATE /ENTRY:mainCRTStartup)
    # The generated code calls boot loops that never return, so what follows
    # them is unreachable (C4702) in a handful of functions: harmless
    set_source_files_properties(${GEN_SOURCES} PROPERTIES COMPILE_OPTIONS /wd4702)
elseif(WIN32)
    # clang or GCC for Windows (llvm-mingw, MinGW-w64). As MSVC builds it:
    # a*b+c stays two operations (the game's floating point must not change),
    # memory of any type can be read as any other and integers wrap around
    target_compile_options(SegaRally2 PRIVATE -ffp-contract=off -fno-strict-aliasing -fwrapv)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        # The generated files contain many functions (clang needs no flag for it)
        target_compile_options(SegaRally2 PRIVATE -Wa,-mbig-obj)
    endif()
endif()
